cmake_minimum_required(VERSION 3.22)

project(Sockets)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Mikoto Options
set(MIKOTO_USE_XCB_WINDOW OFF CACHE BOOL "Disable XCB windowing" FORCE)
set(MIKOTO_USE_WIN32_WINDOW ON CACHE BOOL "Use Win32 windowing" FORCE)

file(GLOB SOURCE_FILES ./*.cc)

add_executable(${PROJECT_NAME} ${SOURCE_FILES})

# Link the engine
target_link_libraries(${PROJECT_NAME} PRIVATE Mikoto)

# Editor-specific includes
target_include_directories(${PROJECT_NAME} PUBLIC .)

target_compile_definitions(${PROJECT_NAME} PRIVATE USE_GLFW_INPUT)
target_compile_definitions(${PROJECT_NAME} PRIVATE SPDLOG_FMT_EXTERNAL)

# The perspective projection matrix generated by GLM will use the OpenGL depth
# range of -1.0 to 1.0 by default. We need to configure it to use the Vulkan
# range of 0.0 to 1.0 using the GLM_FORCE_DEPTH_ZERO_TO_ONE definition.
target_compile_definitions(${PROJECT_NAME} PRIVATE GLM_FORCE_DEPTH_ZERO_TO_ONE)
target_compile_definitions(${PROJECT_NAME} PRIVATE GLM_FORCE_RADIANS)
target_compile_definitions(${PROJECT_NAME} PRIVATE GLM_ENABLE_EXPERIMENTAL)

# ======================================
# Post build (Copy Engine Core Resources)
# ======================================

# Path to your resources folder (relative to project root)
set(RESOURCE_DIR "${CMAKE_SOURCE_DIR}/Mikoto/Resources")

# Path where resources should be copied in the build folder
set(OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}")

add_custom_command(
        TARGET ${PROJECT_NAME} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${RESOURCE_DIR}" "${OUTPUT_DIR}"
        COMMENT "Copying resources to build directory..."
)
